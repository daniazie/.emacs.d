      (interactive)
      (let* ((mb (minibuffer-contents-no-properties))

        (cond ((string-match-p "^[/~:]" lc) (self-insert-command 1 ?/))

      "Move to next candidate in minibuffer, even when minibuffer isn't selected."

    (defun up-from-outside ()
      "Move to previous candidate in minibuffer, even when minibuffer isn't selected."
      (interactive)
      (with-selected-window (active-minibuffer-window)
        (execute-kbd-macro [up])))

    (defun to-and-fro-minibuffer ()
      "Go back and forth between minibuffer and other window."

        (select-window (active-minibuffer-window))))

    (let ((pos (org-find-entry-with-id (my/gh-format-link repo entry id))))
  		 (goto-char pos)
  		 (org-todo)
  		 (let ((prop-state (my/org-get-property "STATE")))
  		   (unless (string= prop-state state))
  		       (progn (org-set-property "STATE" state)
  			      (unless (org-find-exact-headline-in-buffer (capitalize state))

  				(org-insert-heading nil nil 1)

  	 (state (alist-get 'state results))
