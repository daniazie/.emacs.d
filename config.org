  (defun my/convert-time (&optional DATE TIME ZONE)
    (encode-time (parse-time-string (format "%sT%s%s" (format-time-string "%Y-%m-%d" DATE) TIME ZONE))))

  (defun my/check-time-is-new-day ()
    (let ((is_night (time-less-p nil (my/convert-time nil "10:00:00" "+09"))))
      (if (not (member (format-time-string "%A") '("Saturday" "Sunday")))
  	(and (not is_night) (time-less-p nil (my/convert-time nil "15:00:00" "+09")))
        (not is_night))))

#+begin_src emacs-lisp
  (defun my/goto-or-create-timestamp-header ()
    (interactive)
    (when (my/check-time-is-new-day)
      (let* ((timestamp (format-time-string "[%Y-%m-%d %a]"))
  	   (pos (org-find-exact-headline-in-buffer timestamp)))
        (if (not pos)
  	  (progn (goto-char (buffer-size))
  		 (org-insert-heading nil nil t)
  		 (insert timestamp)
  		 (goto-char (buffer-size))
  		 (save-buffer))
  	(goto-char pos))))
    (save-buffer))

  (defun my/open-org-inbox ()
    (interactive)
    (split-window-below)
    (set-buffer
     (get-buffer (find-file (concat org-directory "inbox.org"))))
    (my/goto-or-create-timestamp-header)
    (current-buffer))

  (defun my/open-org-daily ()
    (interactive)
    (select-window (winum-get-window-by-number 2))
    (my/split-window-right)
    (set-buffer
     (get-buffer (find-file (concat org-directory "daily.org"))))
    (when (my/check-time-is-new-day)
      (let* ((timestamp (format-time-string "[%Y-%m-%d %a]"))
  	   (pos (org-find-exact-headline-in-buffer timestamp)))
        (unless pos
  	  (org-capture nil "d"))))
    (save-buffer)
    (current-buffer))

    (interactive)
    (let* ((link (elfeed-entry-link elfeed-show-entry))
  	 (match-idx (string-match "arxiv.org/abs/\\([0-9.]*\\)" link))
           (matched-arxiv-number (match-string 1 link))

        (message "Going to arXiv: %s" matched-arxiv-number)

                        (setq last-arxiv-key key)
                        (setq last-arxiv-title title)
                        (save-buffer)
                        )))
        (save-window-excursion
          (find-file (concat org-directory "papers.org"))
          (goto-char (point-max))
          (insert (format "** TODO Read paper (cite:%s) %s" last-arxiv-key last-arxiv-title))

        (find-file (concat org-directory "agentic-memory/" (format "gh-%ss.org" entry)))

  	  (my/gh-insert-new-entry entry repo id title body author state created-at)
  	(my/gh-update-entry-state entry repo id state))
        (save-buffer))
    nil)

  (defun my/gh-parse-metadata (repo entry results)
    (let* ((title (alist-get 'title results))
  	 (body (alist-get 'body results))
  	 (author (alist-get 'login (alist-get 'author results)))
  	 (state (alist-get 'state results))
  	 (created-date (alist-get 'createdAt results))
  	 (id (car (last (string-split (alist-get 'url results) "/")))))
      (list entry repo id title body author state created-date)))

  (defun my/org-get-property (PROP)
    (interactive)
    (let ((pos (point)))
      (org-entry-get pos PROP)))

  (defun my/gh-fetch-update (repo get-type)
    (let* ((fields "title,body,author,state,createdAt,url")
  	 (cmd (format "gh %s list --repo %s --json %s --state all"
  		      (shell-quote-argument get-type)
  		      (shell-quote-argument repo)
  		      (shell-quote-argument fields))))
      (let* ((output (shell-command-to-string cmd)))
        (unless (string-search "error" output)
  	(let ((results (json-parse-string output :array-type 'list :object-type 'alist)))
           (dolist (result results)
  	   (unless (not result)
  	    (let ((metadata (my/gh-parse-metadata repo get-type result)))
  	      (apply #'my/gh-update metadata)))))))))

  (defun my/gh-fetch-update-continuous ()
    (interactive)
    (let* ((repo (my/gh-select-repo))
  	 (entry (completing-read "Fetch: " '("pr" "issue") nil t)))
      (run-at-time "15 min" 900 #'my/gh-fetch-update repo entry)))

  (keymap-global-set "C-c g f" #'my/gh-fetch-update-continuous)
